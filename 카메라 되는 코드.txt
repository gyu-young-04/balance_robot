#include <M5Unified.h>
#include <esp_camera.h>
#include <Wire.h>

class GC0308 {
private:
public:
    camera_fb_t* fb;
    sensor_t* sensor;
    camera_config_t* config;
    bool begin();
    bool get();
    bool free();
};

static camera_config_t camera_config = {
    .pin_pwdn     = -1,
    .pin_reset    = -1,
    .pin_xclk     = -1,
    .pin_sscb_sda = 12,
    .pin_sscb_scl = 11,
    .pin_d7       = 47,
    .pin_d6       = 48,
    .pin_d5       = 16,
    .pin_d4       = 15,
    .pin_d3       = 42,
    .pin_d2       = 41,
    .pin_d1       = 40,
    .pin_d0       = 39,

    .pin_vsync = 46,
    .pin_href  = 38,
    .pin_pclk  = 45,

    .xclk_freq_hz = 20000000,
    .ledc_timer   = LEDC_TIMER_0,
    .ledc_channel = LEDC_CHANNEL_0,

    .pixel_format  = PIXFORMAT_RGB565,
    .frame_size    = FRAMESIZE_QVGA,
    .jpeg_quality  = 0,
    .fb_count      = 1,
    .fb_location   = CAMERA_FB_IN_DRAM,
    .grab_mode     = CAMERA_GRAB_WHEN_EMPTY,
    .sccb_i2c_port = -1,
};

bool GC0308::begin()
{
    M5.In_I2C.release();
    esp_err_t err = esp_camera_init(&camera_config);
    if (err != ESP_OK) {
        return false;
    }
    sensor = esp_camera_sensor_get();
    return true;
}

bool GC0308::get()
{
    fb = esp_camera_fb_get();
    if (!fb) {
        return false;
    }
    return true;
}

bool GC0308::free()
{
    if (fb) {
        esp_camera_fb_return(fb);
        return true;
    }
    return false;
}

GC0308 Cam;

void setup() {
    auto cfg = M5.config();
    M5.begin(cfg);

    if (!Cam.begin()) {
        Serial.println("Camera Init Fail");
    }
    Serial.println("Camera Init Success");

    Cam.sensor->set_framesize(Cam.sensor, FRAMESIZE_QVGA);
}

void loop() {
     if (Cam.get()) {
        // 디스플레이 크기 대신 카메라 실제 프레임 크기 사용
        M5.Display.pushImage(0, 0, 
                             Cam.fb->width,    // ✅ 실제 프레임 너비
                             Cam.fb->height,   // ✅ 실제 프레임 높이
                             (uint16_t *)Cam.fb->buf);
        Cam.free();
    }
}
