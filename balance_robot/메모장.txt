#include "M5Unified.h"
#include "M5Module4EncoderMotor.h"

M5Module4EncoderMotor driver;

#define LEFT_MOTOR  2
#define RIGHT_MOTOR 3
#define LEFT_DIR    (1)
#define RIGHT_DIR   (-1)
#define LEFT_ENC_DIR  (1)
#define RIGHT_ENC_DIR (1)

float Kp = 15.0;
float Ki = 0.3;    // 추가
float Kd = 0.55;
float Ks = -0.07;
#define PWM_LIMIT 80

float target = -2.0;  // 수직 기준 상대각도, 조정 필요
float g_pitch = 0;
int g_pwm = 0;
float g_speed = 0;

void PIDtask(void* pv) {
    float pitch = 0;
    float integral = 0;
    int32_t prevEncL = 0, prevEncR = 0;
    unsigned long lastTime = micros();
    static float filtered_speed = 0;

    while (1) {
        unsigned long now = micros();
        float dt = (now - lastTime) / 1000000.0;
        lastTime = now;
        if (dt <= 0 || dt > 0.05) { vTaskDelay(1); continue; }

        float ax, ay, az, gx, gy, gz;
        M5.Imu.getAccel(&ax, &ay, &az);
        M5.Imu.getGyro(&gx, &gy, &gz);

        // 참고 코드 방식으로 각도 계산
        float pitchAcc = atan2(az, -ax) * 180.0 / PI;
        pitch = 0.985 * (pitch + (-gy) * dt) + 0.015 * pitchAcc;

        // 엔코더 속도 계산
        int32_t encL = LEFT_ENC_DIR * driver.getEncoderValue(LEFT_MOTOR);
        int32_t encR = RIGHT_ENC_DIR * driver.getEncoderValue(RIGHT_MOTOR);
        float raw_speed = ((encL - prevEncL) + (encR - prevEncR)) / 2.0 / dt;
        prevEncL = encL;
        prevEncR = encR;
        filtered_speed = 0.8 * filtered_speed + 0.2 * raw_speed;

        float error = pitch - target;

        // 자이로 직접 사용한 Kd (더 안정적)
        float pwm_f = (-Kp * error) + (Kd * gy) + (-Ks * filtered_speed);

        int motor_pwm = 0;
        if (abs(pwm_f) > 0.5) {
            motor_pwm = (pwm_f > 0) ? (int)pwm_f + 3 : (int)pwm_f - 3;
        }
        motor_pwm = constrain(motor_pwm, -PWM_LIMIT, PWM_LIMIT);

        // 45도 이상 기울면 모터 끔
        if (abs(pitch) > 45) motor_pwm = 0;

        driver.setMotorSpeed(LEFT_MOTOR, LEFT_DIR * motor_pwm);
        driver.setMotorSpeed(RIGHT_MOTOR, RIGHT_DIR * motor_pwm);

        g_pitch = pitch;
        g_pwm = motor_pwm;
        g_speed = filtered_speed;

        vTaskDelay(pdMS_TO_TICKS(5));
    }
}

void setup() {
    auto cfg = M5.config();
    M5.begin(cfg);
    Serial.begin(115200);
    M5.Imu.begin();
    Wire1.setClock(400000);

    if (!driver.begin(&Wire1, MODULE_4ENCODERMOTOR_ADDR, 11, 12)) {
        Serial.println("Motor Module Error");
        while (1);
    }
    driver.setMode(LEFT_MOTOR, NORMAL_MODE);
    driver.setMode(RIGHT_MOTOR, NORMAL_MODE);

    xTaskCreatePinnedToCore(PIDtask, "PIDtask", 4096, NULL, 5, NULL, 1);
}

void loop() {
    Serial.printf("Pitch:%.2f | PWM:%d | Speed:%.2f | Target:%.2f\n",
                  g_pitch, g_pwm, g_speed, target);
    delay(100);
}

#include "M5Unified.h"
#include "M5Module4EncoderMotor.h"
M5Module4EncoderMotor driver;
#define LEFT_MOTOR  2
#define RIGHT_MOTOR 3
#define LEFT_DIR    (1)
#define RIGHT_DIR   (-1)
#define LEFT_ENC_DIR  (1)
#define RIGHT_ENC_DIR (1)
float Kp = 11.0;
float Ki = 0.1;    // 0으로 시작, 점점 올리기
float Kd = 0.6;
float Ks = -0.23;
#define PWM_LIMIT 80
float target = -1.7;
float g_pitch = 0;
int g_pwm = 0;
float g_speed = 0;
float left_trim = 1.0;
float right_trim = 1.02;  // 직접 튜닝
void PIDtask(void* pv) {
    float pitch = 0;
    float integral = 0;
    int32_t prevEncL = 0, prevEncR = 0;
    unsigned long lastTime = micros();
    static float filtered_speed = 0;
    while (1) {
        unsigned long now = micros();
        float dt = (now - lastTime) / 1000000.0;
        lastTime = now;
        if (dt <= 0 || dt > 0.05) { vTaskDelay(1); continue; }
        float ax, ay, az, gx, gy, gz;
        M5.Imu.getAccel(&ax, &ay, &az);
        M5.Imu.getGyro(&gx, &gy, &gz);
        float pitchAcc = atan2(az, -ax) * 180.0 / PI;
        pitch = 0.985 * (pitch + (-gy) * dt) + 0.015 * pitchAcc;
        int32_t encL = LEFT_ENC_DIR * driver.getEncoderValue(LEFT_MOTOR);
        int32_t encR = RIGHT_ENC_DIR * driver.getEncoderValue(RIGHT_MOTOR);
        float raw_speed = ((encL - prevEncL) + (encR - prevEncR)) / 2.0 / dt;
        prevEncL = encL;
        prevEncR = encR;
        filtered_speed = 0.8 * filtered_speed + 0.2 * raw_speed;
        float error = pitch - target;

        // Ki 추가 (0이면 기존 동작과 완전히 동일)
        integral += error * dt;
        integral = constrain(integral, -5, 5);

        float pwm_f = (-Kp * error) + (-Ki * integral) + (Kd * gy) + (-Ks * filtered_speed);

        int motor_pwm = 0;
        if (abs(pwm_f) > 0.5) {
            motor_pwm = (pwm_f > 0) ? (int)pwm_f + 3 : (int)pwm_f - 3;
        }
        motor_pwm = constrain(motor_pwm, -PWM_LIMIT, PWM_LIMIT);
        if (abs(pitch) > 45) {
            motor_pwm = 0;
            integral = 0;  // 쓰러지면 적분 초기화
        }
        driver.setMotorSpeed(LEFT_MOTOR, LEFT_DIR * motor_pwm * left_trim);
        driver.setMotorSpeed(RIGHT_MOTOR, RIGHT_DIR * motor_pwm * right_trim);
        g_pitch = pitch;
        g_pwm = motor_pwm;
        g_speed = filtered_speed;
        vTaskDelay(pdMS_TO_TICKS(5));
    }
}
void setup() {
    auto cfg = M5.config();
    M5.begin(cfg);
    Serial.begin(115200);
    M5.Imu.begin();
    Wire1.setClock(400000);
    if (!driver.begin(&Wire1, MODULE_4ENCODERMOTOR_ADDR, 11, 12)) {
        Serial.println("Motor Module Error");
        while (1);
    }
    driver.setMode(LEFT_MOTOR, NORMAL_MODE);
    driver.setMode(RIGHT_MOTOR, NORMAL_MODE);
    xTaskCreatePinnedToCore(PIDtask, "PIDtask", 4096, NULL, 5, NULL, 1);
}
void loop() {
    Serial.printf("Pitch:%.2f | PWM:%d | Speed:%.2f | Target:%.2f\n",
                  g_pitch, g_pwm, g_speed, target);
    delay(100);
}
