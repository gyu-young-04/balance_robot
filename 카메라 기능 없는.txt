#include <M5Unified.h>
#include <MicroTFLite.h>
#include "esp_camera.h"
#include "model_data.h"

const int CANVAS_SIZE      = 160;
const int MODEL_INPUT_SIZE = 28;
const int ARENA_SIZE       = 96 * 1024;

alignas(16) uint8_t tensor_arena[ARENA_SIZE];

M5Canvas canvas(&M5.Display);

bool isCameraMode = false;
bool drawing      = false;

int lastX = -1;
int lastY = -1;

void runInference();
void preprocessTouch();
void preprocessCamera();
bool initCamera();

// =====================================================
// 카메라 초기화
// =====================================================

bool initCamera() {
    camera_config_t config;

    config.ledc_channel = LEDC_CHANNEL_0;
    config.ledc_timer   = LEDC_TIMER_0;
    config.pin_d0 = 11; config.pin_d1 = 9;
    config.pin_d2 = 8;  config.pin_d3 = 10;
    config.pin_d4 = 12; config.pin_d5 = 18;
    config.pin_d6 = 17; config.pin_d7 = 16;
    config.pin_xclk     = 15;
    config.pin_pclk     = 13;
    config.pin_vsync    = 6;
    config.pin_href     = 7;
    config.pin_sccb_sda = 12;
    config.pin_sccb_scl = 11;
    config.pin_pwdn     = -1;
    config.pin_reset    = -1;
    config.xclk_freq_hz = 20000000;
    config.pixel_format = PIXFORMAT_RGB565;
    config.frame_size   = FRAMESIZE_QVGA;
    config.fb_location  = CAMERA_FB_IN_PSRAM;
    config.jpeg_quality = 12;
    config.fb_count     = 2;

    esp_err_t err = esp_camera_init(&config);
    if (err != ESP_OK) {
        Serial.printf("❌ Camera Init Failed : 0x%x\n", err);
        return false;
    }
    Serial.println("✅ Camera Init Success");
    return true;
}

// =====================================================
// 브러시
// =====================================================

void drawBrush(int cx, int cy) {
    for (int by = -5; by <= 5; by++) {
        for (int bx = -5; bx <= 5; bx++) {
            float dist = sqrtf(bx * bx + by * by);
            if (dist <= 5.0f) {
                int px = cx + bx;
                int py = cy + by;
                if (px >= 0 && px < CANVAS_SIZE &&
                    py >= 0 && py < CANVAS_SIZE) {

                    uint16_t existing = canvas.readPixel(px, py);
                    uint8_t existingBright = (existing >> 8) & 0xF8;
                    uint8_t newBright = (uint8_t)(255.0f * (1.0f - dist / 5.0f));
                    uint8_t finalBright = max(existingBright, newBright);

                    canvas.drawPixel(px, py,
                        canvas.color565(finalBright, finalBright, finalBright));
                }
            }
        }
    }
}

// =====================================================
// setup
// =====================================================

void setup() {
    auto cfg = M5.config();
    cfg.internal_imu = true;
    M5.begin(cfg);

    Serial.begin(115200);
    delay(500);

    if (!ModelInit(model_tflite, tensor_arena, ARENA_SIZE)) {
        Serial.println("❌ Model Init Failed!");
        while (1);
    }
    Serial.println("✅ Model Init Success!");

    //initCamera();

    M5.Display.setRotation(1);
    M5.Display.fillScreen(BLACK);

    canvas.createSprite(CANVAS_SIZE, CANVAS_SIZE);
    canvas.fillSprite(BLACK);
    canvas.pushSprite(80, 40);

    M5.Display.setTextColor(WHITE);
    M5.Display.setTextDatum(middle_center);
    M5.Display.drawString("Draw Number",       160, 20);
    M5.Display.drawString("A: Mode   C: Clear", 160, 225);
}

// =====================================================
// loop
// =====================================================

void loop() {
    M5.update();
    auto detail = M5.Touch.getDetail();

    bool isBottomTouched = detail.isPressed() && (detail.y > 200);

    // A : MODE 전환
    if (M5.BtnA.wasPressed() ||
        (isBottomTouched && detail.x < 100)) {

        isCameraMode = !isCameraMode;
        lastX = -1; lastY = -1; drawing = false;

        M5.Display.fillScreen(BLACK);
        M5.Display.setTextColor(WHITE);
        M5.Display.setTextDatum(middle_center);
        M5.Display.drawString("A: Mode   C: Clear", 160, 225);

        if (isCameraMode) {
            M5.Display.drawString("CAMERA MODE", 160, 20);
        } else {
            M5.Display.drawString("TOUCH MODE",  160, 20);
            canvas.pushSprite(80, 40);
        }

        delay(200);
        return;
    }

    // C : CLEAR
    if (M5.BtnC.wasPressed() ||
        (isBottomTouched && detail.x > 220)) {

        canvas.fillSprite(BLACK);
        canvas.pushSprite(80, 40);

        M5.Display.fillRect(0, 190, 320, 50, BLACK);

        lastX = -1; lastY = -1; drawing = false;

        Serial.println(">>> Canvas Cleared");
        delay(200);
        return;
    }

    // =================================================
    // TOUCH MODE
    // =================================================

    if (!isCameraMode) {

        if (detail.isPressed() && detail.y < 200) {
            drawing = true;

            int tx = detail.x - 80;
            int ty = detail.y - 40;

            if (tx >= 0 && tx < CANVAS_SIZE &&
                ty >= 0 && ty < CANVAS_SIZE) {

                if (lastX >= 0 && lastY >= 0) {
                    int dx = tx - lastX;
                    int dy = ty - lastY;
                    int steps = max(abs(dx), abs(dy));
                    if (steps < 1) steps = 1;

                    for (int i = 0; i <= steps; i++) {
                        float t = (float)i / steps;
                        drawBrush(lastX + dx * t, lastY + dy * t);
                    }
                } else {
                    drawBrush(tx, ty);
                }

                lastX = tx; lastY = ty;
                canvas.pushSprite(80, 40);
            }
        }

        // 손 뗐을 때 자동 추론
        if (!detail.isPressed() && drawing) {
            drawing = false;
            lastX = -1; lastY = -1;
            preprocessTouch();
            runInference();
        }

    }

    // =================================================
    // CAMERA MODE
    // =================================================

    else {
        camera_fb_t *fb = esp_camera_fb_get();
        if (fb) {
            M5.Display.pushImage(40, 40, fb->width, fb->height, (uint16_t*)fb->buf);
            esp_camera_fb_return(fb);
        }

        // B : 캡처 후 추론
        if (M5.BtnB.wasPressed()) {
            preprocessCamera();
            runInference();
        }
    }
}

// =====================================================
// TOUCH 전처리
// =====================================================

void preprocessTouch() {
    Serial.println("\n--- Input Tensor Visual (28x28) ---");

    float raw[28][28];
    float blockSize = (float)CANVAS_SIZE / 28.0f;

    for (int y = 0; y < 28; y++) {
        for (int x = 0; x < 28; x++) {
            int x0 = (int)(x * blockSize),     y0 = (int)(y * blockSize);
            int x1 = (int)((x+1) * blockSize), y1 = (int)((y+1) * blockSize);
            float maxBright = 0.0f;

            for (int sy = y0; sy < y1; sy++) {
                for (int sx = x0; sx < x1; sx++) {
                    uint16_t color = canvas.readPixel(sx, sy);
                    float bright = ((float)((color >> 11) & 0x1F) / 31.0f +
                                    (float)((color >>  5) & 0x3F) / 63.0f +
                                    (float)( color        & 0x1F) / 31.0f) / 3.0f;
                    if (bright > maxBright) maxBright = bright;
                }
            }
            raw[y][x] = maxBright;
        }
    }

    int minX = 28, minY = 28, maxX = 0, maxY = 0;
    for (int y = 0; y < 28; y++) {
        for (int x = 0; x < 28; x++) {
            if (raw[y][x] > 0.18f) {
                if (x < minX) minX = x; if (x > maxX) maxX = x;
                if (y < minY) minY = y; if (y > maxY) maxY = y;
            }
        }
    }

    if (maxX <= minX || maxY <= minY) {
        for (int i = 0; i < 28*28; i++) ModelSetInput(0.0f, i);
        Serial.println("No Drawing");
        return;
    }

    float normalized[28][28] = {0};
    int   boxW    = maxX - minX + 1;
    int   boxH    = maxY - minY + 1;
    float scale   = 20.0f / max(boxW, boxH);
    int   scaledW = (int)(boxW * scale);
    int   scaledH = (int)(boxH * scale);
    int   offsetX = (28 - scaledW) / 2;
    int   offsetY = (28 - scaledH) / 2;

    for (int y = 0; y < scaledH; y++) {
        for (int x = 0; x < scaledW; x++) {
            int srcX = minX + (int)(x / scale);
            int srcY = minY + (int)(y / scale);
            if (srcX >= 0 && srcX < 28 && srcY >= 0 && srcY < 28)
                normalized[y + offsetY][x + offsetX] = raw[srcY][srcX];
        }
    }

    for (int y = 0; y < 28; y++) {
        for (int x = 0; x < 28; x++) {
            float v = normalized[y][x];
            ModelSetInput(v, y * 28 + x);
            Serial.print(v > 0.18f ? "O" : ".");
        }
        Serial.println();
    }
}

// =====================================================
// CAMERA 전처리
// =====================================================

void preprocessCamera() {
    camera_fb_t *fb = esp_camera_fb_get();
    if (!fb) return;

    uint16_t *pixels = (uint16_t*)fb->buf;
    int camW = fb->width;
    int camH = fb->height;

    for (int y = 0; y < 28; y++) {
        for (int x = 0; x < 28; x++) {
            int srcX = map(x, 0, 27, 0, camW - 1);
            int srcY = map(y, 0, 27, 0, camH - 1);
            uint16_t color = pixels[srcY * camW + srcX];

            float bright = ((float)((color >> 11) & 0x1F) / 31.0f +
                            (float)((color >>  5) & 0x3F) / 63.0f +
                            (float)( color        & 0x1F) / 31.0f) / 3.0f;

            float value = 1.0f - bright;
            if (value < 0.2f) value = 0.0f;

            ModelSetInput(value, y * 28 + x);
        }
    }

    esp_camera_fb_return(fb);
}

// =====================================================
// 추론
// =====================================================

void runInference() {
    if (!ModelRunInference()) {
        Serial.println("Inference Failed!");
        return;
    }

    float max_score = -1.0f;
    int   prediction = -1;

    for (int i = 0; i < 10; i++) {
        float score = ModelGetOutput(i);
        Serial.printf("%d : %.5f\n", i, score);
        if (score > max_score) { max_score = score; prediction = i; }
    }

    M5.Display.fillRect(0, 190, 320, 40, BLACK);
    M5.Display.setTextColor(YELLOW);
    M5.Display.setTextSize(2);
    M5.Display.setCursor(60, 200);

    if (max_score > 0.1f) {
        M5.Display.printf("DETECT: %d (%d%%)", prediction, (int)(max_score * 100));
        Serial.printf("\n>>> Predicted: %d  confidence: %.5f\n", prediction, max_score);
    } else {
        M5.Display.printf("DETECT: ?");
    }
}
